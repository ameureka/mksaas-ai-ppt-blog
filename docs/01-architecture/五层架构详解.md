## 架构全景

```
┌────────────────────────────────────────────────────┐
│                用户界面层 (Layer 1)                 │
│  Pages, Layouts, App Router, 用户交互入口          │
└─────────────────────────────────────────────────────┘
                        ↓ 组件组合
┌────────────────────────────────────────────────────┐
│                组件层 (Layer 2)                     │
│  React 组件, 可复用组件库, 展示和容器组件         │
└────────────────────────────────────────────────────┘
                        ↓ 调用
┌────────────────────────────────────────────────────┐
│                服务层 (Layer 3)                     │
│  Server Actions, 业务逻辑, 数据转换               │
└────────────────────────────────────────────────────┘
                        ↓ HTTP
┌────────────────────────────────────────────────────┐
│                API 层 (Layer 4)                     │
│  Next.js API Routes, REST endpoints               │
└────────────────────────────────────────────────────┘
                        ↓ SQL
┌────────────────────────────────────────────────────┐
│                数据层 (Layer 5)                     │
│  PostgreSQL, Drizzle ORM, 数据库表和迁移         │
└────────────────────────────────────────────────────┘
```

## 第三层：服务层

### 职责

- 实现业务逻辑
- 验证数据
- 调用数据层
- 返回类型安全的结果

### Server Actions

```typescript
// src/actions/dashboard.ts
'use server'

import { auth } from '@/lib/auth'
import { db } from '@/db'
import { user, payment } from '@/db/schema'
import { eq } from 'drizzle-orm'

// 类型定义
export type DashboardData = {
  stats: {
    totalUsers: number
    totalRevenue: number
    activeSubscriptions: number
  }
  chartData: Array<{ date: string; value: number }>
}

// 权限检查客户端
import { createSafeActionClient } from 'next-safe-action'

const actionClient = createSafeActionClient()

const userActionClient = actionClient.use(async ({ next }) => {
  const session = await auth()
  if (!session) throw new Error('未授权')

  return next({ ctx: { session } })
})

// Server Action 实现
export const getDashboardData = userActionClient.action(async ({ ctx }) => {
  const userId = ctx.session.user.id

  // 获取用户统计
  const userStats = await db.query.user.findFirst({
    where: eq(user.id, userId),
  })

  // 获取支付信息
  const payments = await db.query.payment.findMany({
    where: eq(payment.userId, userId),
  })

  // 返回类型安全的数据
  return {
    stats: {
      totalUsers: userStats?.totalCredits || 0,
      totalRevenue: payments.reduce((sum, p) => sum + p.amount, 0),
      activeSubscriptions: payments.filter((p) => p.status === 'active').length,
    },
    chartData: [], // 生成图表数据
  } satisfies DashboardData
})
```

### Server Actions 的三层权限

```typescript
// 1. actionClient - 公开操作 (无需认证)
const publicAction = actionClient.schema(schema).action(async ({ parsedInput }) => {
  // 例: 提交联系表单
  return { success: true }
})

// 2. userActionClient - 用户操作 (需要登录)
const userAction = userActionClient.schema(schema).action(async ({ parsedInput, ctx }) => {
  // ctx.session 保证存在
  const userId = ctx.session.user.id
  // 例: 更新用户资料
  return { success: true }
})

// 3. adminActionClient - 管理员操作 (需要管理员权限)
const adminAction = adminActionClient
  .schema(schema)
  .action(async ({ parsedInput, ctx }) => {
    // 验证管理员权限
    if (ctx.session.user.role !== 'admin') throw new Error('禁止')
    // 例: 删除用户
    return { success: true }
  })
```

### 验证和错误处理

```typescript
import { z } from 'zod'

const UpdateProfileSchema = z.object({
  name: z.string().min(2).max(100),
  email: z.string().email(),
  bio: z.string().max(500).optional(),
})

export const updateProfile = userActionClient
  .schema(UpdateProfileSchema)
  .action(async ({ parsedInput, ctx }) => {
    try {
      // parsedInput 已通过 Zod 验证，类型安全
      const { name, email, bio } = parsedInput
      const userId = ctx.session.user.id

      // 检查邮箱唯一性
      const existing = await db.query.user.findFirst({
        where: eq(user.email, email),
      })

      if (existing && existing.id !== userId) {
        return { error: '邮箱已被使用' }
      }

      // 更新数据库
      await db.update(user).set({ name, email, bio }).where(eq(user.id, userId))

      return { success: true, data: { name, email } }
    } catch (error) {
      console.error('更新资料失败:', error)
      return { error: '更新失败，请重试' }
    }
  })
```

### Next.js API Routes

```typescript
// src/app/api/webhooks/stripe/route.ts
import { stripe } from '@/lib/stripe'
import { handleStripeCheckoutCompleted } from '@/services/payment'

export async function POST(request: Request) {
  const body = await request.text()
  const signature = request.headers.get('stripe-signature')

  // 验证 Stripe 签名
  const event = stripe.webhooks.constructEvent(
    body,
    signature!,
    process.env.STRIPE_WEBHOOK_SECRET!
  )

  switch (event.type) {
    case 'checkout.session.completed':
      await handleStripeCheckoutCompleted(event.data.object)
      break
    // 处理其他事件
  }

  return Response.json({ received: true })
}
```

### REST API 示例

```typescript
// src/app/api/users/[id]/route.ts
import { db } from '@/db'
import { user } from '@/db/schema'
import { eq } from 'drizzle-orm'

export async function GET(request: Request, { params }: { params: { id: string } }) {
  const userData = await db.query.user.findFirst({
    where: eq(user.id, params.id),
  })

  if (!userData) {
    return Response.json({ error: '用户不存在' }, { status: 404 })
  }

  return Response.json(userData)
}

export async function PUT(request: Request, { params }: { params: { id: string } }) {
  const data = await request.json()

  await db.update(user).set(data).where(eq(user.id, params.id))

  return Response.json({ success: true })
}
```

## 第五层：数据层

### 职责

- 定义数据模式
- 执行数据库操作

- 索引和优化

### Drizzle Schema

```typescript
// src/db/schema.ts
import { pgTable, text, timestamp, integer, boolean } from 'drizzle-orm/pg-core'
import { sql } from 'drizzle-orm'

export const user = pgTable('user', {
  id: text('id').primaryKey().default(sql`uuid_generate_v4()`),
  email: text('email').notNull().unique(),
  name: text('name').notNull(),
  password: text('password'), // nullable for OAuth users
  tier: text('tier').default('free'), // 'free', 'pro', 'lifetime'
  totalCredits: integer('total_credits').default(0),
  emailVerified: boolean('email_verified').default(false),
  createdAt: timestamp('created_at').defaultNow(),
  updatedAt: timestamp('updated_at').defaultNow(),
})

export const payment = pgTable('payment', {
  id: text('id').primaryKey().default(sql`uuid_generate_v4()`),
  userId: text('user_id')
    .notNull()
    .references(() => user.id, { onDelete: 'cascade' }),
  stripeId: text('stripe_id').unique(),
  amount: integer('amount').notNull(), // 美分
  status: text('status').default('pending'), // 'pending', 'completed', 'failed'
  tier: text('tier').notNull(), // 'pro', 'lifetime'
  createdAt: timestamp('created_at').defaultNow(),
})
```

## 层级间通信

### 数据从下而上流动 (Query)

```
Component (第2层)
    ↓ 调用
Server Action (第3层)
    ↓ 调用
Drizzle Query (第5层)
    ↓ 返回 SQL 结果
Server Action
    ↓ 返回 TypeScript 数据
Component
    ↓ 渲染 UI
```

### 命令从上而下流动 (Mutation)

```
User Action (第1层)
    ↓
React Event Handler (第2层)
    ↓
Server Action Call (第3层)
    ↓ 验证
Drizzle Operation (第5层)
    ↓ 数据库更新
Server Action 返回 (第3层)
    ↓
UI 重新渲染 (第2层)
```

## 类型系统设计

### 贯穿所有层的类型

```typescript
// 1. 数据库 Schema 类型
type User = typeof user.$inferSelect
type NewUser = typeof user.$inferInsert

type UserResponse = {
  id: string
  email: string
  name: string
  tier: 'free' | 'pro' | 'lifetime'

type UpdateProfileInput = z.infer<typeof UpdateProfileSchema>
type UpdateProfileResult = { success: true } | { error: string }

// 4. Component Props 类型
interface UserCardProps {
  user: UserResponse
  onUpdate?: (user: UserResponse) => void
}

// 5. Hook 返回类型
type UseUserReturn = {
  user: UserResponse | null
  isLoading: boolean
```

```typescript
  return (
    <form action={formAction}>
      <input type="email" name="email" required />

// 3. 服务层：业务逻辑
// src/actions/auth.ts
import { z } from 'zod'
import { db } from '@/db'
import { user } from '@/db/schema'
import { createSafeActionClient } from 'next-safe-action'

const RegisterSchema = z.object({
  email: z.string().email(),
  password: z.string().min(8),
})

const actionClient = createSafeActionClient()

export const registerUser = actionClient.schema(RegisterSchema).action(
  async ({ parsedInput }) => {
    const { email, password } = parsedInput

    // 检查邮箱是否已存在
    const existing = await db.query.user.findFirst({
      where: (users, { eq }) => eq(users.email, email),
    })

    if (existing) {
      return { error: '邮箱已注册' }
    }

    // 创建用户
    const hashedPassword = await hashPassword(password)
    const newUser = await db.insert(user).values({
      email,
      password: hashedPassword,
      name: email.split('@')[0],
    })

    // 发送验证邮件
    await sendVerificationEmail(email)

    return { success: true, userId: newUser.insertId }
  }
)

// 4. 数据层：数据库定义
```

## 总结

五层架构的好处:

✅ **关注点分离** - 每层只做一件事
✅ **可维护性** - 改动局限在特定层
✅ **可测试性** - 每层可独立测试
✅ **可复用性** - 组件和逻辑可复用
✅ **类型安全** - 类型贯穿整个流程
✅ **易于扩展** - 新功能可按层添加
