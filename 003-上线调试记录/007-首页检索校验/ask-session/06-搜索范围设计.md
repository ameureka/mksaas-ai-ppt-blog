## 一、向量搜索范围

向量生成输入 (已在 02-pgvector配置决策.md 确认):

```typescript
const generateEmbeddingInput = (ppt: PPT): string => {
  const parts = [
    ppt.title,
    ppt.description || '',
    (ppt.tags || []).join(' '),
  ];
  return parts.filter(Boolean).join(' ');
};
```

## 二、SQL 降级搜索范围

### 2.1 修改前

```typescript
// 当前实现
if (params?.search?.trim()) {
  const keyword = `%${params.search.trim()}%`;
  conditions.push(
    or(
      ilike(pptTable.title, keyword),
      ilike(pptTable.author, keyword),
      sql`array_to_string(${pptTable.tags}, ',') ILIKE ${keyword}`
    )
  );
}
```

### 2.2 修改后

```typescript
// 加入 description
if (params?.search?.trim()) {
```
