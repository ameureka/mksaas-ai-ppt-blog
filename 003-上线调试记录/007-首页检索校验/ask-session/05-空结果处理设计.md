### 2.2 编辑距离算法

```typescript
// 计算两个字符串的编辑距离 (Levenshtein Distance)
function levenshteinDistance(a: string, b: string): number {
  const matrix: number[][] = [];
  
  for (let i = 0; i <= a.length; i++) {
    matrix[i] = [i];
  }
  for (let j = 0; j <= b.length; j++) {
    matrix[0][j] = j;
  }
  
  for (let i = 1; i <= a.length; i++) {
    for (let j = 1; j <= b.length; j++) {
      const cost = a[i - 1] === b[j - 1] ? 0 : 1;
      matrix[i][j] = Math.min(
        matrix[i - 1][j] + 1,      // 删除
        matrix[i][j - 1] + 1,      // 插入
        matrix[i - 1][j - 1] + cost // 替换
      );
    }
  }
  
  return matrix[a.length][b.length];
}
```

### 3.2 后端实现

```typescript
// src/actions/search/corrections.ts

// 获取纠错建议
export async function getSearchCorrections(query: string): Promise<string[]> {
  const db = await getDb();
  const results: { word: string; score: number }[] = [];
  
  // 1. 热词模糊匹配 (编辑距离)
  const hotWords = await db
    .select({ keyword: hotKeywords.keyword })
    .from(hotKeywords);
  
  for (const hw of hotWords) {
    const distance = levenshteinDistance(query, hw.keyword);
    if (distance <= 2 && distance > 0) {
      results.push({ word: hw.keyword, score: 100 - distance * 20 });
    }
  }
  
  // 2. 向量相似度 (如果模糊匹配结果不足3个)
  if (results.length < 3) {
    try {
      const queryEmbedding = await generateEmbedding(query);
      
      const similar = await db.execute(sql`
        SELECT keyword, 1 - (embedding <=> ${queryEmbedding}) as similarity
        FROM hot_keywords
        WHERE embedding IS NOT NULL
        ORDER BY embedding <=> ${queryEmbedding}
        LIMIT 5
      `);
      
      for (const s of similar.rows) {
        if (s.similarity > 0.7 && s.keyword !== query) {
          results.push({ word: s.keyword, score: s.similarity * 50 });
        }
      }
    } catch (error) {
      console.error('[Corrections] Vector search failed:', error);
    }
  }
  
  // 去重 + 排序 + 取 Top 3
  const seen = new Set<string>();
  const unique = results
    .filter(r => {
      if (seen.has(r.word)) return false;
      seen.add(r.word);
      return true;
    })
    .sort((a, b) => b.score - a.score);
  
  return unique.slice(0, 3).map(r => r.word);
}

// 获取热门推荐 PPT
export async function getHotPPTs(limit: number = 15): Promise<PPT[]> {
  const db = await getDb();
  
  const rows = await db
    .select()
    .from(pptTable)
    .where(isNull(pptTable.deletedAt))
    .orderBy(desc(pptTable.downloadCount))
    .limit(limit);
  
  return rows.map(toPPTDto);
}
```

### 3.3 搜索 API 集成

```typescript
// src/app/api/ppts/route.ts
export async function GET(req: NextRequest) {
  // ... 现有搜索逻辑
  
  const result = await getPPTs(params);
  
  // 如果结果为空，添加建议
  if (result.success && result.data.items.length === 0 && params.search) {
    const [corrections, hotPPTs] = await Promise.all([
      getSearchCorrections(params.search),
      getHotPPTs(15),
    ]);
    
    return Response.json({
      ...result,
      suggestions: {
        corrections,
        hotPPTs,
      },
    });
  }
  
  return Response.json(result);
}
```

## 四、前端组件

```typescript
// components/search/EmptySearchResult.tsx
interface Props {
  query: string;
  corrections: string[];
  hotPPTs: PPT[];
  onCorrectionClick: (keyword: string) => void;
  onDownload: (ppt: PPT) => void;
}

export function EmptySearchResult({ 
  query, 
  corrections, 
  hotPPTs, 
  onCorrectionClick,
  onDownload,
}: Props) {
  return (
    <div className="py-12">
      {/* 未找到提示 */}
      <div className="text-center mb-8">
        <div className="inline-flex items-center justify-center w-16 h-16 rounded-full bg-muted mb-4">
          <Search className="w-8 h-8 text-muted-foreground" />
        </div>
        <p className="text-lg text-muted-foreground">
          未找到 "<span className="font-medium text-foreground">{query}</span>" 相关结果
        </p>
      </div>
      
      {/* 纠错建议 */}
      {corrections.length > 0 && (
        <div className="mb-8 text-center">
          <p className="text-sm text-muted-foreground mb-3">
            💡 您是否想搜索：
          </p>
          <div className="flex justify-center flex-wrap gap-2">
            {corrections.map((word, i) => (
              <button
                key={i}
                type="button"
                onClick={() => onCorrectionClick(word)}
                className="px-4 py-2 rounded-full bg-primary/10 text-primary hover:bg-primary/20 transition-colors"
              >
                {word}
              </button>
            ))}
          </div>
        </div>
      )}
      
      {/* 热门推荐 */}
      {hotPPTs.length > 0 && (
        <div>
          <h3 className="text-xl font-bold mb-6 flex items-center gap-2">
            <Flame className="w-5 h-5 text-orange-500" />
            热门推荐
          </h3>
          <div className="grid gap-6 sm:grid-cols-2 lg:grid-cols-3 xl:grid-cols-5">
            {hotPPTs.map(ppt => (
              <PPTCard 
                key={ppt.id} 
                ppt={ppt} 
                onDownload={onDownload}
              />
            ))}
          </div>
        </div>
      )}
    </div>
  );
}
```

## 五、使用示例

```typescript
// 在搜索页面中使用
const handleSearch = async (query: string) => {
  const res = await fetch(`/api/ppts?search=${encodeURIComponent(query)}`);
  const json = await res.json();
  
  if (json.success) {
    setResults(json.data.items);
    
    // 处理空结果
    if (json.data.items.length === 0 && json.suggestions) {
      setCorrections(json.suggestions.corrections);
      setHotPPTs(json.suggestions.hotPPTs);
      setShowEmptyState(true);
    } else {
      setShowEmptyState(false);
    }
  }
};

// 渲染
{showEmptyState ? (
  <EmptySearchResult
    query={query}
    corrections={corrections}
    hotPPTs={hotPPTs}
    onCorrectionClick={(word) => {
      setQuery(word);
      handleSearch(word);
    }}
    onDownload={handleDownload}
  />
) : (
  // 正常搜索结果
)}
```
