## 完整 search_log 表设计

```sql
CREATE TABLE search_log (
  id TEXT PRIMARY KEY,
  
  -- 用户关联
  user_id TEXT REFERENCES "user"(id) ON DELETE SET NULL,
  
  -- 搜索内容
  keyword TEXT NOT NULL,
  
  -- 搜索结果
  result_count INTEGER DEFAULT 0,
  clicked_ppt_id TEXT,
  
  -- 搜索来源
  source TEXT DEFAULT 'search',           -- search, hot_keyword, suggestion
  from_suggestion BOOLEAN DEFAULT FALSE,  -- 是否来自建议词点击
  
  -- 搜索性能
  search_type TEXT,                       -- vector, sql, hybrid
  duration_ms INTEGER,                    -- 搜索耗时(毫秒)
  
  -- 设备信息
  ip_address TEXT,
  user_agent TEXT,
  
  -- 时间
  created_at TIMESTAMP DEFAULT NOW()
);

-- 索引
CREATE INDEX search_log_keyword_idx ON search_log(keyword);
CREATE INDEX search_log_user_id_idx ON search_log(user_id);
CREATE INDEX search_log_created_at_idx ON search_log(created_at);
CREATE INDEX search_log_search_type_idx ON search_log(search_type);
```

## 埋点记录逻辑

```typescript
// src/actions/search/log.ts
interface SearchLogInput {
  keyword: string;
  userId?: string;
  resultCount: number;
  source?: 'search' | 'hot_keyword' | 'suggestion';
  fromSuggestion?: boolean;
  searchType: 'vector' | 'sql' | 'hybrid';
  durationMs: number;
  ipAddress?: string;
  userAgent?: string;
}

export async function logSearch(input: SearchLogInput) {
  const db = await getDb();
  
  await db.insert(searchLog).values({
    id: `sl_${randomUUID()}`,
    keyword: input.keyword,
    userId: input.userId,
    resultCount: input.resultCount,
    source: input.source || 'search',
    fromSuggestion: input.fromSuggestion || false,
    searchType: input.searchType,
    durationMs: input.durationMs,
    ipAddress: input.ipAddress,
    userAgent: input.userAgent,
    createdAt: new Date(),
  });
}

// 记录点击行为
export async function logSearchClick(searchLogId: string, pptId: string) {
  const db = await getDb();
  
  await db
    .update(searchLog)
    .set({ clickedPptId: pptId })
    .where(eq(searchLog.id, searchLogId));
}
```

## 搜索 API 集成

```typescript
// src/app/api/ppts/route.ts
export async function GET(req: NextRequest) {
  const startTime = Date.now();
  
  // ... 搜索逻辑
  const result = await searchPPTs(params);
  
  const durationMs = Date.now() - startTime;
  
  // 记录埋点 (异步，不阻塞响应)
  if (params.search) {
    logSearch({
      keyword: params.search,
      userId: session?.user?.id,
      resultCount: result.data.total,
      source: params.source,
      fromSuggestion: params.fromSuggestion === 'true',
      searchType: result.searchType, // 'vector' | 'sql' | 'hybrid'
      durationMs,
      ipAddress: req.headers.get('x-forwarded-for') || undefined,
      userAgent: req.headers.get('user-agent') || undefined,
    }).catch(console.error);
  }
  
  return Response.json(result);
}
```

## 可分析的指标

| 指标 | SQL | 用途 |
|------|-----|------|
| 热门搜索词 | `GROUP BY keyword ORDER BY COUNT(*) DESC` | 热词展示 |
| 搜索转化率 | `clicked_ppt_id IS NOT NULL` 比例 | 优化搜索质量 |
| 零结果率 | `WHERE result_count = 0` 比例 | 发现内容缺口 |
| 平均耗时 | `AVG(duration_ms)` | 性能监控 |
| 向量搜索占比 | `WHERE search_type = 'vector'` 比例 | 技术分析 |
| 建议词点击率 | `WHERE from_suggestion = true` 比例 | 优化建议功能 |
