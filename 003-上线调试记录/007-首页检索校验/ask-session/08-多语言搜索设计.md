## 一、向量搜索多语言支持

### 1.1 天然支持

BGE 模型 (BAAI/bge-large-zh-v1.5) 特点：
- ✅ 中文优化，繁简体语义相近
- ✅ 中英文混合文本支持
- ✅ "PPT模板" 与 "PowerPoint Template" 语义相近

```
搜索 "年終總結" (繁体)
  ↓ 生成向量
  ↓ 与 "年终总结" (简体) 的向量相似度高
  ↓ 返回相关结果 ✅
```

### 1.2 无需额外处理

向量搜索不需要繁简转换，模型已处理语义相似性。

## 二、SQL 降级繁简转换

### 2.1 方案

SQL 搜索时，同时搜索繁体和简体版本：

```typescript
import { traditionalToSimplified, simplifiedToTraditional } from '@/lib/chinese-convert';

function buildSearchConditions(keyword: string) {
  const simplified = traditionalToSimplified(keyword);
  const traditional = simplifiedToTraditional(keyword);
  
  // 去重
  const variants = [...new Set([keyword, simplified, traditional])];
  
  // 构建 OR 条件
  const conditions = variants.flatMap(k => [
    ilike(pptTable.title, `%${k}%`),
    ilike(pptTable.description, `%${k}%`),
  ]);
  
  return or(...conditions);
}
```

### 2.2 繁简转换库

推荐使用轻量级库 `opencc-js`:

```bash
pnpm add opencc-js
```

```typescript
// src/lib/chinese-convert.ts
import OpenCC from 'opencc-js';

const converterT2S = OpenCC.Converter({ from: 'tw', to: 'cn' });
const converterS2T = OpenCC.Converter({ from: 'cn', to: 'tw' });

export function traditionalToSimplified(text: string): string {
  return converterT2S(text);
}

export function simplifiedToTraditional(text: string): string {
  return converterS2T(text);
}
```

### 2.3 修改 buildWhere 函数

```typescript
// src/actions/ppt/ppt.ts
import { traditionalToSimplified, simplifiedToTraditional } from '@/lib/chinese-convert';

const buildWhere = (params?: PPTListParams) => {
  const conditions = [];
  conditions.push(isNull(pptTable.deletedAt));

  if (params?.search?.trim()) {
    const keyword = params.search.trim();
    
    // 繁简体变体
    const simplified = traditionalToSimplified(keyword);
    const traditional = simplifiedToTraditional(keyword);
    const variants = [...new Set([keyword, simplified, traditional])];
    
    // 构建搜索条件
    const searchConditions = variants.flatMap(k => {
      const pattern = `%${k}%`;
      return [
        ilike(pptTable.title, pattern),
        ilike(pptTable.description, pattern),
        ilike(pptTable.author, pattern),
        sql`array_to_string(${pptTable.tags}, ',') ILIKE ${pattern}`,
      ];
    });
    
    conditions.push(or(...searchConditions));
  }

  // ... 其他条件
  return and(...conditions);
};
```
