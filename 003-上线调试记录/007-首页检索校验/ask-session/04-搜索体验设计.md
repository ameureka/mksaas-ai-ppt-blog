```typescript
// hooks/useDebounce.ts
import { useState, useEffect } from 'react';

export function useDebounce<T>(value: T, delay: number): T {
  const [debouncedValue, setDebouncedValue] = useState<T>(value);

  useEffect(() => {
    const timer = setTimeout(() => setDebouncedValue(value), delay);
    return () => clearTimeout(timer);
  }, [value, delay]);

  return debouncedValue;
}

// 使用示例
const [query, setQuery] = useState('');
const debouncedQuery = useDebounce(query, 300);

useEffect(() => {
  if (debouncedQuery.trim()) {
    fetchSuggestions(debouncedQuery);
  }
}, [debouncedQuery]);
```

### 2.3 localStorage 操作

```typescript
// lib/search-history.ts
const STORAGE_KEY = 'ppt_search_history';
const MAX_LOCAL_HISTORY = 10;

export function getLocalSearchHistory(): string[] {
  if (typeof window === 'undefined') return [];
  const data = localStorage.getItem(STORAGE_KEY);
  return data ? JSON.parse(data) : [];
}

export function addLocalSearchHistory(keyword: string): void {
  if (typeof window === 'undefined') return;
  const history = getLocalSearchHistory();
  
  // 去重，移到最前
  const filtered = history.filter(h => h !== keyword);
  const updated = [keyword, ...filtered].slice(0, MAX_LOCAL_HISTORY);
  
  localStorage.setItem(STORAGE_KEY, JSON.stringify(updated));
}

export function clearLocalSearchHistory(): void {
  if (typeof window === 'undefined') return;
  localStorage.removeItem(STORAGE_KEY);
}
```

### 2.4 登录后合并

```typescript
// 用户登录后调用
async function syncSearchHistoryOnLogin(userId: string) {
  const localHistory = getLocalSearchHistory();
  
  if (localHistory.length > 0) {
    // 批量写入数据库
    await fetch('/api/search/sync-history', {
      method: 'POST',
      body: JSON.stringify({ keywords: localHistory }),
    });
    
    // 清空本地
    clearLocalSearchHistory();
  }
}
```

### 3.3 后端实现

```typescript
// src/app/api/search/suggestions/route.ts
export async function GET(req: NextRequest) {
  const { searchParams } = new URL(req.url);
  const query = searchParams.get('q')?.trim();
  
  if (!query || query.length < 1) {
    return Response.json({ suggestions: [] });
  }
  
  const session = await getSession();
  const userId = session?.user?.id;
  
  const suggestions = await getSuggestions(query, userId);
  return Response.json({ suggestions });
}

async function getSuggestions(query: string, userId?: string) {
  const db = await getDb();
  const results: { text: string; source: string; priority: number }[] = [];
  
  // 1. 用户搜索历史 (优先级 100)
  if (userId) {
    const history = await db
      .selectDistinct({ keyword: searchLog.keyword })
      .from(searchLog)
      .where(and(
        eq(searchLog.userId, userId),
        ilike(searchLog.keyword, `${query}%`)
      ))
      .orderBy(desc(searchLog.createdAt))
      .limit(5);
    
    history.forEach(h => results.push({ 
      text: h.keyword, 
      source: 'history', 
      priority: 100 
    }));
  }
  
  // 2. 热门关键词 (优先级 50)
  const hotWords = await db
    .select({ keyword: hotKeywords.keyword })
    .from(hotKeywords)
    .where(ilike(hotKeywords.keyword, `%${query}%`))
    .limit(3);
  
  hotWords.forEach(h => results.push({ 
    text: h.keyword, 
    source: 'hot', 
    priority: 50 
  }));
  
  // 3. PPT 标题匹配 (优先级 10)
  const titles = await db
    .selectDistinct({ title: pptTable.title })
    .from(pptTable)
    .where(and(
      isNull(pptTable.deletedAt),
      ilike(pptTable.title, `%${query}%`)
    ))
    .limit(5);
  
  titles.forEach(t => results.push({ 
    text: t.title, 
    source: 'title', 
    priority: 10 
  }));
  
  // 去重 + 排序 + 取 Top 8
  const seen = new Set<string>();
  const unique = results.filter(r => {
    const lower = r.text.toLowerCase();
    if (seen.has(lower)) return false;
    seen.add(lower);
    return true;
  });
  
  return unique
    .sort((a, b) => b.priority - a.priority)
    .slice(0, 8)
    .map(r => ({ text: r.text, source: r.source }));
}
```

### 3.5 前端组件

```typescript
// components/search/SearchSuggestions.tsx
interface Props {
  query: string;
  onSelect: (keyword: string) => void;
}

export function SearchSuggestions({ query, onSelect }: Props) {
  const [suggestions, setSuggestions] = useState<SuggestionItem[]>([]);
  const debouncedQuery = useDebounce(query, 300);
  
  useEffect(() => {
    if (!debouncedQuery.trim()) {
      setSuggestions([]);
      return;
    }
    
    fetch(`/api/search/suggestions?q=${encodeURIComponent(debouncedQuery)}`)
      .then(res => res.json())
      .then(data => setSuggestions(data.suggestions));
  }, [debouncedQuery]);
  
  if (suggestions.length === 0) return null;
  
  const icons = {
    history: '🕐',
    hot: '🔥',
    title: '📄',
  };
  
  return (
    <div className="absolute top-full left-0 right-0 bg-background border rounded-lg shadow-lg mt-1 z-50">
      {suggestions.map((item, index) => (
        <button
          key={index}
          className="w-full px-4 py-2 text-left hover:bg-muted flex items-center gap-2"
          onClick={() => onSelect(item.text)}
        >
          <span>{icons[item.source]}</span>
          <span>{item.text}</span>
        </button>
      ))}
    </div>
  );
}
```
