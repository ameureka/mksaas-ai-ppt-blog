修复 4: 分离三个板块的数据请求
useEffect(() => {
  const fetchCategoryPPTs = async () => {
    setIsLoading(true);
    try {
      // 并行请求三种数据
      const [hotRes, newRes, allRes] = await Promise.all([
        fetch(`/api/ppts?category=${slug}&pageSize=8&sortBy=downloads&sortOrder=desc`),
        fetch(`/api/ppts?category=${slug}&pageSize=8&sortBy=created_at&sortOrder=desc`),
        fetch(`/api/ppts?category=${slug}&page=${page}&pageSize=12&sortBy=${sortBy === 'newest' ? 'created_at' : 'downloads'}&sortOrder=desc`),
      ]);

      const [hotJson, newJson, allJson] = await Promise.all([
        hotRes.json(), newRes.json(), allRes.json()
      ]);

      if (hotJson.success) setHotPPTs(mapItems(hotJson.data.items));
      if (newJson.success) setNewPPTs(mapItems(newJson.data.items));
      // featuredPPTs 可以用 is_featured 字段筛选，或暂时用 hotPPTs
      setFeaturedPPTs(hotJson.success ? mapItems(hotJson.data.items) : []);

      if (allJson.success) {
        setAllPPTs(mapItems(allJson.data.items));
        setTotalPages(Math.max(1, Math.ceil((allJson.data.total ?? 0) / 12)));
      }
    } catch (error) {
      toast.error('加载分类数据失败');
    } finally {
      setIsLoading(false);
    }
  };
  fetchCategoryPPTs();
}, [slug, sortBy, page]);

三、问题优先级排序
| 优先级 | 问题 | 影响范围 | 修复难度 | |--------|------|----------|----------| | P0 | slug 映射不完整 | 4个分类页面异常 | 低 | | P0 | categoryMetadata 缺失 | 7个分类显示错误元数据 | 中 | | P1 | 相关分类跳转错误 | 所有分类页 | 低 | | P1 | 三板块数据相同 | 用户体验 | 中 | | P1 | 排序参数未生效 | 用户体验 | 低 | | P2 | 统计数据硬编码 | 数据准确性 | 低(暂不改) | | P2 | SEO canonical 问题 | SEO | 低 | | P2 | 分页总数计算 | 显示准确性 | 低 |

四、建议修复顺序
修复 slugToName 映射 - 从 PPT_CATEGORIES 自动生成
补全 categoryMetadata - 添加缺失的 7 个分类
修复相关分类跳转 - 使用 slug 而非中文名
分离板块数据请求 - 热门/精选/最新使用不同排序
修复排序参数 - sortBy 变化时使用对应排序
修复 SEO canonical - 使用 slug
