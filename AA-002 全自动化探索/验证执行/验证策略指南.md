### 3. 集成测试脚本

创建测试脚本 `scripts/test-search-system.ts`:

```typescript
import { db } from '@/db';
import { ppt, searchLog } from '@/db/schema';
import { generateEmbedding } from '@/lib/embedding';
import { hybridSearch } from '@/actions/ppt/search';

async function testSearchSystem() {
  console.log('🧪 开始搜索系统测试...\n');

  // 1. 测试向量生成
  console.log('1️⃣ 测试向量生成...');
  try {
    const embedding = await generateEmbedding('测试内容');
    console.log('✅ 向量生成成功，维度:', embedding.length);
  } catch (error) {
    console.error('❌ 向量生成失败:', error);
  }

  // 2. 测试混合搜索
  console.log('\n2️⃣ 测试混合搜索...');
  try {
    const results = await hybridSearch({
      query: '商业计划书',
      limit: 5
    });
    console.log('✅ 搜索返回结果数:', results.length);
  } catch (error) {
    console.error('❌ 搜索失败:', error);
  }

  // 3. 测试搜索日志
  console.log('\n3️⃣ 测试搜索日志...');
  try {
    const logs = await db.select().from(searchLog).limit(5);
    console.log('✅ 搜索日志记录数:', logs.length);
  } catch (error) {
    console.error('❌ 日志查询失败:', error);
  }

  console.log('\n✨ 测试完成！');
}

testSearchSystem();
```

运行：`tsx scripts/test-search-system.ts`

### 4. 性能测试

创建性能测试脚本 `scripts/benchmark-search.ts`:

```typescript
import { performance } from 'perf_hooks';

async function benchmarkSearch() {
  const queries = [
    '商业计划书',
    'business plan',
    '年终总结PPT',
    '教育培训模板',
    'marketing presentation'
  ];

  for (const query of queries) {
    const start = performance.now();
    
    const response = await fetch('http://localhost:3005/api/search/ppt', {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify({ query, hybrid: true })
    });
    
    const data = await response.json();
    const end = performance.now();
    
    console.log(`查询: "${query}"`);
    console.log(`  结果数: ${data.results?.length || 0}`);
    console.log(`  耗时: ${(end - start).toFixed(2)}ms\n`);
  }
}

benchmarkSearch();
```
